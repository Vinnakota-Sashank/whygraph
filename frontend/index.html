# Generated with WhyGraph MCP Guardrail (FalkorDB)
import os
import json
import redis

# WhyGraph Guard: Enforcing ADR-001 (Shared Redis Cluster & Connection Pooling)
REDIS_POOL = redis.ConnectionPool(
    host=os.getenv('REDIS_HOST', 'redis-cluster.internal'),
    port=6379,
    max_connections=20,  # ADR-001: Pool max 20 connections
    decode_responses=True
)

class CartCache:
    def __init__(self):
        self.client = redis.Redis(connection_pool=REDIS_POOL)

    def set_cart(self, user_id: str, items: list):
        # ADR-001: Mandated namespace 'checkout:cart:{id}' & explicit 3600s TTL
        key = f'checkout:cart:{user_id}'
        self.client.setex(key, 3600, json.dumps(items))

    def get_cart(self, user_id: str):
        key = f'checkout:cart:{user_id}'
        data = self.client.get(key)
        return json.loads(data) if data else None
🛡️ Active Guardrails Injected via FalkorDB MCP:
  • ADR-001: MUST use shared Redis cluster with connection pool (5-20 connections).
  • ADR-001: Cache key format {service}:{entity}:{id}.
  • ADR-001: Mandatory explicit TTL expiration on all keys.
  • Verified in sub-millisecond OpenCypher traversal against live FalkorDB Cloud.
